## Конфигурация подключения MySQL
## Используется для статистики, сбора отзывов,
## администрирования и внутриигровой библиотеки.

## Должен ли SQL быть включён? Раскомментируйте для включения
SQL_ENABLED

## Сервер, на котором находится база данных MySQL.
## Примеры: localhost, 200.135.5.43, www.mysqldb.com и т.д.
ADDRESS localhost

## Порт сервера MySQL (по умолчанию 3306).
PORT 3306

## База данных для всех функций SQL, не только для отзывов.
FEEDBACK_DATABASE tgstation_schema

## Префикс, добавляемый к названию каждой таблицы. Для старых баз данных потребуется установить значение erro_
## Примечание: это не изменяет названия таблиц в базе данных, вам придётся сделать это самостоятельно.
##НАПРИМЕР:
##	FEEDBACK_TABLEPREFIX
##	FEEDBACK_TABLEPREFIX SS13_
## Уберите "SS13_", если вы используете стандартный файл схемы.
FEEDBACK_TABLEPREFIX

## Имя пользователя/Логин для доступа к базе данных.
FEEDBACK_LOGIN root

## Пароль для доступа к базе данных.
FEEDBACK_PASSWORD root

## Время в секундах до тайм-аута асинхронных запросов
## Установите 0 для бесконечности
ASYNC_QUERY_TIMEOUT 10

## Время в секундах выполнения блокирующих запросов до срабатывания тайм-аута медленного запроса
## Установите 0 для бесконечности
## Должно быть меньше или равно ASYNC_QUERY_TIMEOUT
BLOCKING_QUERY_TIMEOUT 5

## Минимальное количество SQL-соединений, поддерживаемых в пуле. Увеличение этого значения на серверах, географически удалённых от базы данных, может улучшить производительность.
POOLING_MIN_SQL_CONNECTIONS 1

## Максимальное количество SQL-соединений с базой данных.
POOLING_MAX_SQL_CONNECTIONS 25

## Максимальное количество одновременных асинхронных запросов, которые могут ожидать результата, прежде чем мы начнём ставить дальнейшие запросы к базе данных в очередь.
MAX_CONCURRENT_QUERIES 25
